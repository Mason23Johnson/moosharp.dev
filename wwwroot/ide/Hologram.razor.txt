    protected override async Task OnAfterRenderAsync(bool firstRender)
    {
        if (firstRender)
        {
            await JS.InvokeVoidAsync("hologramLab.init", "hologramStage", "hologramRig");
        }
    }

    private void OpenTerminalFrame()
    {
        screenMode = "terminal";
        _ = StartTerminalBootAsync();
    }

    private void OpenDrive(string drive)
    {
        CancelTerminalBoot();
        currentDrive = drive;
        explorerSubfolder = null;
        screenMode = "explorer";
        openedFileEntry = null;
    }

    private void OpenExplorerApp()
    {
        CancelTerminalBoot();
        currentDrive = RootDrive;
        explorerSubfolder = null;
        screenMode = "explorer";
        openedFileEntry = null;
    }

    private void NavigateUp()
    {
        if (!string.IsNullOrEmpty(explorerSubfolder))
        {
            explorerSubfolder = null;
            openedFileEntry = null;
            return;
        }

        if (currentDrive != RootDrive)
        {
            currentDrive = RootDrive;
            openedFileEntry = null;
            return;
        }

        ReturnToDesktop();
    }

    private void ReturnToDesktop()
    {
        CancelTerminalBoot();
        screenMode = "desktop";
        openedFileEntry = null;
    }

    private void ActivateExplorerEntry(HoloExplorerEntry entry)
    {
        if (entry.Id is "hdrive" or "gdrive")
        {
            OpenDrive(entry.Id);
            return;
        }

        if (entry.Kind == "folder")
        {
            explorerSubfolder = entry.Id;
            openedFileEntry = null;
            return;
        }

        openedFileEntry = entry;
    }

    private void CloseOpenedFile()
    {
        openedFileEntry = null;
    }

    private async Task StartTerminalBootAsync()
    {
        CancelTerminalBoot(disposeOnly: true);
        terminalBootCts = new CancellationTokenSource();
        var token = terminalBootCts.Token;

        terminalOutput = string.Empty;
        terminalCommand = string.Empty;
        terminalReady = false;
        await InvokeAsync(StateHasChanged);

        try
        {
            foreach (var line in terminalBootLines)
            {
                foreach (var character in line)
                {
                    terminalOutput += character;
                    await InvokeAsync(StateHasChanged);
                    await Task.Delay(18, token);
                }

                terminalOutput += "\n";
                await InvokeAsync(StateHasChanged);
                await Task.Delay(140, token);
            }

            terminalReady = true;
            await InvokeAsync(StateHasChanged);
        }
        catch (OperationCanceledException)
        {
        }
    }

    private void SkipTerminalBoot()
    {
        if (terminalReady || screenMode != "terminal")
        {
            return;
        }

        terminalBootCts?.Cancel();
        terminalOutput = string.Join("\n", terminalBootLines) + "\n";
        terminalReady = true;
    }

    private async Task HandleTerminalKeydown(KeyboardEventArgs args)
    {
        if (args.Key != "Enter" || !terminalReady)
        {
            return;
        }

        await ExecuteTerminalCommandAsync();
    }

    private async Task ExecuteTerminalCommandAsync()
    {
        var command = terminalCommand.Trim();
        if (string.IsNullOrWhiteSpace(command))
        {
            return;
        }

        terminalOutput += $"> {command}\n";

        switch (command.ToLowerInvariant())
        {
            case "/help":
                terminalOutput += "/help  /home  /about  /projects  /contact  /arcade\n";
                terminalOutput += "/desktop  /hdrive  /gdrive  /clear\n";
                break;
            case "/home":
                Navigation.NavigateTo("/");
                return;
            case "/about":
                Navigation.NavigateTo("/about");
                return;
            case "/projects":
                Navigation.NavigateTo("/projects");
                return;
            case "/contact":
                Navigation.NavigateTo("/contact");
                return;
            case "/arcade":
                terminalOutput += ">> SYSTEM: Redirecting back to Modern for Moochine Arcade\n";
                terminalOutput += ">> SYSTEM: Redirecting to Moochine Arcade...\n";
                terminalCommand = string.Empty;
                await InvokeAsync(StateHasChanged);
                await Task.Delay(4000);
                Navigation.NavigateTo("/arcade");
                return;
            case "/desktop":
                terminalOutput += "Returning to desktop...\n";
                ReturnToDesktop();
                break;
            case "/hdrive":
            case "/h":
                terminalOutput += "Opening H: MooVault...\n";
                OpenDrive("hdrive");
                break;
            case "/gdrive":
            case "/g":
                terminalOutput += "Opening G: Personal...\n";
                OpenDrive("gdrive");
                break;
            case "/clear":
                terminalOutput = string.Empty;
                break;
            default:
                terminalOutput += $"Unknown command: {command}\n";
                break;
        }

        terminalCommand = string.Empty;
    }

    private void CancelTerminalBoot(bool disposeOnly = false)
    {
        if (terminalBootCts is null)
        {
            return;
        }

        terminalBootCts.Cancel();
        terminalBootCts.Dispose();
        terminalBootCts = null;

        if (!disposeOnly)
        {
            terminalReady = false;
            terminalCommand = string.Empty;
        }
    }

    public void Dispose()
    {
        CancelTerminalBoot(disposeOnly: true);
    }

    private sealed record HoloExplorerEntry(string Id, string Title, string Description, string Kind, string Glyph);
}
